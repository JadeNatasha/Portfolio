class CNN(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=(3,3), stride=1, padding=1)
        self.pool1 = nn.MaxPool2d(kernel_size=(2,2), stride=2, padding=1)

        self.conv2 = nn.Conv2d(32, 64, kernel_size=(3,3), stride=1, padding=1)
        self.conv3 = nn.Conv2d(64, 128, kernel_size=(3,3), stride=1, padding=1)
        self.conv4 = nn.Conv2d(128, 256, kernel_size=(3,3), stride=1, padding=1)
        self.pool2 = nn.MaxPool2d(kernel_size=(2,2), stride=2, padding=0)

        self.dropout = nn.Dropout(p=0.2)

        self.fcn1 = nn.Linear(256*7*7, 1000)
        self.fcn2 = nn.Linear(1000, 500)
        self.fcn3 = nn.Linear(500, num_classes)

    def forward(self, x):
        x = F.relu(self.conv1(x))
        x = self.pool1(x)

        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
        x = F.relu(self.conv4(x))
        x = self.pool2(x)

        x = self.dropout(x)
        x = x.view(-1, 256*7*7)

        x = F.relu(self.fcn1(x))
        x = F.relu(self.fcn2(x))
        x = F.softmax(self.fcn3(x), dim=1)

        return x

model = CNN(num_classes=10)

This creates an actual instance of the model and prints its structure, so we can visually confirm every layer got built with the right shapes.

dum = torch.randn(4, 1, 28, 28)
out = model(dum)
print(out.shape)

torch.Size([4, 10])

Feeds 4 fake, randomly-generated “images” through the model just to test that every layer connects correctly end-to-end — no real data or training needed yet. Getting [4, 10] back means: 4 images went in, and for each one, the model produced 10 numbers.

This exercise reinforced the importance of manually verifying tensor shapes at each stage of a CNN, especially since diagrams don’t always explicitly state every parameter (like channel counts here), requiring us to cross-reference given information and compute the rest.
